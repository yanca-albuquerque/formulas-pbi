Html_Tabela_Faltas = VAR TabelaMeses = SUMMARIZE( Calendario, Calendario[NumMes], Calendario[Mes] ) -- 1. Cabeçalho (Jan, Fev, Mar...) VAR CabecalhoMeses = CONCATENATEX( TabelaMeses, VAR MesAbrev = UPPER(LEFT(Calendario[Mes], 3)) RETURN "" & MesAbrev & "", "", Calendario[NumMes], ASC ) -- 2. Linha 1: Total de Faltas VAR Linha1_Valores = CONCATENATEX( TabelaMeses, VAR MesNum = Calendario[NumMes] VAR Val = CALCULATE([Total Dias de Faltas], Calendario[NumMes] = MesNum) RETURN "" & COALESCE(FORMAT(Val, "#,##0"), "0") & "", "", Calendario[NumMes], ASC ) -- 3. Linha 2: Colaboradores Envolvidos VAR Linha2_Valores = CONCATENATEX( TabelaMeses, VAR MesNum = Calendario[NumMes] VAR Val = CALCULATE([Total de Colaboradores Faltas], Calendario[NumMes] = MesNum) RETURN "" & COALESCE(FORMAT(Val, "#,##0"), "0") & "", "", Calendario[NumMes], ASC ) -- 4. Linha 3: Unidade com maior incidência (quebra de linha permitida em cada coluna) VAR Linha3_Valores = CONCATENATEX( TabelaMeses, VAR MesNum = Calendario[NumMes] VAR Val = CALCULATE([Top1_Secao_Faltas], Calendario[NumMes] = MesNum) RETURN "" & COALESCE(Val, "-") & "", "", Calendario[NumMes], ASC ) RETURN "
" & CabecalhoMeses & " " & Linha1_Valores & " " & Linha2_Valores & " " & Linha3_Valores & "
